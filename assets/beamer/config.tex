% A compact 2--6 item TOC. For longer agendas, split into more than one page.
\newcounter{toccount}
\newcommand{\DrawTOC}[2][1]{%
\begin{frame}[plain]
  \begin{tikzpicture}[remember picture,overlay]
    \fill[gdutred] (current page.south west) rectangle ([xshift=3.7cm]current page.north west);
    \node[text=white,align=center,font=\fontsize{27}{34}\selectfont\bfseries]
      at ([xshift=1.85cm]current page.west) {汇报\\提纲};
    \node[text=white,font=\scriptsize] at ([xshift=1.85cm,yshift=-1.55cm]current page.west) {CONTENTS};
    \setcounter{toccount}{0}\foreach \x in {#2}{\stepcounter{toccount}}
    \pgfmathsetmacro{\tocstep}{min(1.45,6.2/max(1,\thetoccount))}
    \foreach \x [count=\xi] in {#2}{%
      \pgfmathsetmacro{\tocy}{(\thetoccount-1)*\tocstep/2-(\xi-1)*\tocstep}
      \ifnum\xi=#1\relax
        \node[fill=gdutred,text=white,text width=9cm,minimum height=.88cm,align=left,inner xsep=.35cm,font=\large\bfseries]
          at ([xshift=1.7cm,yshift=\tocy cm]current page.center) {\xi\quad\x};
      \else
        \node[fill=gdutpale,text=muted,text width=9cm,minimum height=.88cm,align=left,inner xsep=.35cm,font=\large]
          at ([xshift=1.7cm,yshift=\tocy cm]current page.center) {\xi\quad\x};
      \fi}
  \end{tikzpicture}
\end{frame}}
